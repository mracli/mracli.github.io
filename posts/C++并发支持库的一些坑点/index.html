// std::thread 实际构造函数
template <class _Fn, class... _Args, enable_if_t<!is_same_v<_Remove_cvref_t<_Fn>, thread>, int> = 0>
    _NODISCARD_CTOR_THREAD explicit thread(_Fn&& _Fx, _Args&&... _Ax) {
        _Start(_STD forward<_Fn>(_Fx), _STD forward<_Args>(_Ax)...);
    }

// 其中 _Start 实现如下
template <class _Fn, class... _Args>
void _Start(_Fn&& _Fx, _Args&&... _Ax) {
    using _Tuple                 = tuple<decay_t<_Fn>, decay_t<_Args>...>;
    auto _Decay_copied           = _STD make_unique<_Tuple>(_STD forward<_Fn>(_Fx), _STD forward<_Args>(_Ax)...);
    constexpr auto _Invoker_proc = _Get_invoke<_Tuple>(make_index_sequence<1 + sizeof...(_Args)>{});

    _Thr._Hnd =
        reinterpret_cast<void*>(_CSTD _beginthreadex(nullptr, 0, _Invoker_proc, _Decay_copied.get(), 0, &_Thr._Id));

    if (_Thr._Hnd) { // ownership transferred to the thread
        (void) _Decay_copied.release();
    } else { // failed to start thread
        _Thr._Id = 0;
        _Throw_Cpp_error(_RESOURCE_UNAVAILABLE_TRY_AGAIN);
    }
}

可以看到构造过程中, thread 构造函数 将可调用对象和所有参数完美转发到 _Start 函数中，而在 _Start 中可以看到：

在构建线程的过程中，使用:

1

using _Tuple = tuple<decay_t<_Fn>, decay_t<_Args>...>;

上述类型表示，将可调用对象和所有的参数都预先做了 decay_t 转换处理，将原有的 cv 限定以及引用全部退化掉，使得传入的参数实际都采用的按值传递。

为了解决这中退化问题，对于用户在构建线程传递参数的过程中，应默认所有的参数都为按值传递，为了实现按引用传递，使用 std::ref() 对变量进行包装，实现实际传递引用：

#include <iostream>
#include <thread>

void func(const int &n) {
    std::cout << &n << '\n';
}

int main() {
    int n;
    std::thread t(func, std::ref(n));
    std::cout << &n << '\n';

    if(t.joinable()) {
        t.join();
    }

    return 0;
}
