秋招以来第一道宏相关的笔试题

很简单，但是宏，而且挺有意思，下面是我当时给出的答案。

#include <iostream>

#define __PP_REV_IMPL_0
#define __PP_REV_IMPL_2(str) str
#define __PP_REV_IMPL_4(str) __PP_REV_IMPL_2(str.substr(2)) + str.substr(0, 2)
#define __PP_REV_IMPL_6(str) __PP_REV_IMPL_4(str.substr(2)) + str.substr(0, 2)
#define __PP_REV_IMPL_8(str) __PP_REV_IMPL_6(str.substr(2)) + str.substr(0, 2)

#define __PP_REV(n, str) __PP_REV_IMPL_##n(str)
#define __PP_PRE(str) (str.substr(0, 2))
#define __PP_CALC_REV(str, n) (__PP_PRE(str) + __PP_REV(n, str.substr(2)))

using namespace std;

int main() {
    string tmp1, tmp2;
    while (cin >> tmp1 >> tmp2) {
        // cout << "DEBUG: " << tmp1 << '\n' << tmp2 <<'\n';
        cout << __PP_CALC_REV(tmp1, 8) << ' ' << __PP_CALC_REV(tmp2, 4) << '\n';
    }
    return 0;
}
