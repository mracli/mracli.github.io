标准库-shared_ptr 实现

共享所有权的智能指针

发表于 2024/10/29 更新于 2024/10/29
作者 墨墨鱼
9 分钟阅读

std::shared_ptr 分类为智能指针，目的在于接替上古时期 auto_ptr，同时解决历史遗留问题。本节我们从 shared_ptr 实现的角度出发，尝试深入浅出地构建。

认知

对于 std::shared_ptr 我们应该很清楚，该类可以帮助我们管理具有动态生命周期或者资源管理需求的对象，同时可在多个指针中共享同一对象，也就是：

  1. 以 RAII 的角度管理资源/指针；
  2. 多个 shared_ptr 之间可以安全地共享同一份指针/资源；

那么据此，可以得知 shared_ptr 的实现应该为一种间接拥有式的指针类型，实际使用单独的对象进行管理资源，也就是：

为了达成上述目的：SPBlock(shared_ptr_block) 应该统计当前持有自身的 shared_ptr 数量，并在适当的时候析构自身; 由于每种指针或者资源对象释放自身资源的方式不同，所以对应的释放策略也应该以闭包的形式传入，并支持调用

// 实际管理资源的类, 该类不需要移动
template<class T>
struct SPBlock {
public:
    explicit SPBlock(T *data_ptr) : data_ptr_(data_ptr), ref_(1) {}

    SPBlock(SPBlock &&) = delete;

    void incref() { ref_.fetch_add(1, std::memory_order_relaxed); }

    void decref() {
        if (ref_count_.fetch_sub(1, std::memory_order_relaxed) == 1) {
            delete this;
        }
    }

    ~SPBlock() {
        delete data_ptr_;
    }

public:
    std::atomic<long> ref_;
    T *data_ptr_;
};

// SharedPtr 则通过 SPBlock 间接访问到管理的资源
template<class T>
struct SharedPtr {
    explicit SharedPtr(T *ptr) : block_(new SPBlock<T>(ptr)) {}
    SharedPtr(std::nullptr_t = nullptr) : block_(nullptr) {}

    // 拷贝逻辑，实现多个shared_ptr 之间共享
    SharedPtr(const SharedPtr<T> &that) : block_(that.block_) {
        block_->incref();
    }

    SharedPtr &operator =(const SharedPtr<T> &that) {
        if(this == &that) return *this;
        if(block_ != nullptr) {
            block_->decref();
        }
        block_ = that.block_;
        if(block_ != nullptr) {
            block_->incref();
        }
        return *this;
    }

    // 移动指针则不影响 `ref_count`
    SharedPtr(SharedPtr<T> &&that) : block_(std::exchange(that.block_, nullptr)) {}

    SharedPtr& operator =(SharedPtr<T> &&that) {
        if(this == &that) return *this;
        if(block_ != nullptr) {
            block_->decref();
        }
        block_ = std::exchange(that.block_, nullptr);
        return *this;
    }

public:
    SPBlock<T> *block_;
};

// 关于计数则单独拎出来处理，降低心智成本
struct SPBlock {
public:
    explicit SPBlock() : ref_(1) {}

    SPBlock(SPBlock &&) = delete;

    void incref() { ref_.fetch_add(1, std::memory_order_relaxed); }

    void decref() {
        if (ref_count_.fetch_sub(1, std::memory_order_relaxed) == 1) {
            delete this;
        }
    }

    virtual ~SPBlock() = default;

public:
    std::atomic<long> ref_;
};

template<class T>
struct DefaultDeleter {
    void operator()(T *ptr) {
        delete ptr;
    }
};

// 将删除作为策略单独提出, 也即 Deleter
template <class T, class Deleter = DefaultDeleter<T>>
struct SPBlockImpl : public SPBlock {
public:
    explicit SPBlockImpl(T *ptr) : data_ptr_(ptr) {}
    explicit SPBlockImpl(T *ptr, Deleter &&deleter) : data_ptr_(ptr), deleter_(std::move(deleter)) {}

    ~SPBlockImpl() noexcept override {
        deleter_(data_ptr_);
    }

public:
    T *data_ptr_;
    Deleter deleter_;
};

// 这样我们就开放出 Deleter 的操作，可以在初始化时以函数对象传入
template <class T>
struct SharedPtr {
public:
    explicit SharedPtr(T *ptr) : ptr_(ptr), block_(new SPBlockImpl<T>(ptr)) {}
    template<class Deleter>
    explicit SharedPtr(T *ptr, Deleter del) :ptr_(ptr), block_(new SPBlockImpl<T, Deleter>(ptr, std::move(del))) {}
    SharedPtr(std::nullptr_t = nullptr) : block_(nullptr) {}
    SharedPtr(const SharedPtr<T> &that) : block_(that.block_) {
        block_->ref_inc();
    }

    SharedPtr<T> &operator=(const SharedPtr<T> &that) {
        if (this == &that) return *this;
        if (block_) {
            block_->ref_dec();
        }
        block_ = that.block_;
        if (block_) {
            block_->ref_inc();
        }
        return *this;
    }
    ~SharedPtr() {
        if (block_) {
            block_->ref_dec();
        }
    }

    long use_count() const noexcept {
        return block_ == nullptr ? 0 : block_->use_count();
    }

public:
    T *ptr_; // 为什么要额外增添指针？防止二次寻址(block_->ptr_)，减少取址时间成本
    SPBlock *block_;
};

// Helper Function 包装
template <class T, class... Args>
SharedPtr<T> MakeShared(Args &&...args) {
    return SharedPtr<T>(new T(std::forward<Args>(args)...));
}

template <class T>
struct SharedPtr {
public:
    template <class Y,
              std::enable_if_t<std::is_convertible_v<Y *, T *>, int> = 0>
    explicit SharedPtr(Y *ptr) : ptr_(ptr), block_(new SPBlockImpl<Y>(ptr)) {}

    template <class Y, class Deleter,
              std::enable_if_t<std::is_convertible_v<Y *, T *>, int> = 0>
    explicit SharedPtr(Y *ptr, Deleter del)
        : ptr_(ptr), block_(new SPBlockImpl<Y, Deleter>(ptr, std::move(del))) {}
    SharedPtr(std::nullptr_t = nullptr) : block_(nullptr) {}
    SharedPtr(const SharedPtr<T> &that) : ptr_(that.ptr_), block_(that.block_) {
        block_->ref_inc();
    }
    template <class Y,
              std::enable_if_t<std::is_convertible_v<Y *, T *>, int> = 0>
    SharedPtr(const SharedPtr<T> &that) : ptr_(that.ptr_), block_(that.block_) {
        block_->ref_inc();
    }

    template <class Y>
    SharedPtr(const SharedPtr<Y> &that, T *ptr)
        : ptr_(ptr), block_(that.block_) {
        block_->ref_inc();
    }

    template <class Y>
    SharedPtr(SharedPtr<Y> &&that, T *ptr)
        : ptr_(std::exchange(ptr, nullptr)),
          block_(std::exchange(that.block_, nullptr)) {
        block_->ref_inc();
    }

    SharedPtr &operator=(const SharedPtr<T> &that) {
        if (this == &that) return *this;
        if (block_) {
            block_->ref_dec();
        }
        block_ = that.block_;
        ptr_ = that.ptr_;
        if (block_) {
            block_->ref_inc();
        }
        return *this;
    }
    template <class Y,
              std::enable_if_t<std::is_convertible_v<Y *, T *>, int> = 0>
    SharedPtr &operator=(const SharedPtr<Y> &that) {
        if (this == &that) return *this;
        if (block_) {
            block_->ref_dec();
        }
        block_ = that.block_;
        ptr_ = that.ptr_;
        if (block_) {
            block_->ref_inc();
        }
        return *this;
    }

    SharedPtr(SharedPtr<T> &&that)
        : ptr_(std::exchange(that.ptr_, nullptr)),
          block_(std::exchange(that.block_, nullptr)) {}

    template <class Y,
              std::enable_if_t<std::is_convertible_v<Y *, T *>, int> = 0>
    SharedPtr(SharedPtr<Y> &&that)
        : ptr_(std::exchange(that.ptr_, nullptr)),
          block_(std::exchange(that.block_, nullptr)) {}

    SharedPtr &operator=(SharedPtr<T> &&that) {
        if (this == &that) return *this;
        if (block_) {
            block_->ref_dec();
        }
        ptr_ = std::exchange(that.ptr_, nullptr);
        block_ = std::exchange(that.block_, nullptr);
        return *this;
    }

    template <class Y,
              std::enable_if_t<std::is_convertible_v<Y *, T *>, int> = 0>
    SharedPtr &operator=(SharedPtr<Y> &&that) {
        if (this == &that) return *this;
        if (block_) {
            block_->ref_dec();
        }
        ptr_ = std::exchange(that.ptr_, nullptr);
        block_ = std::exchange(that.block_, nullptr);
        return *this;
    }

    ~SharedPtr() {
        if (block_) {
            block_->ref_dec();
        }
    }

    long use_count() const noexcept {
        return block_ == nullptr ? 0 : block_->use_count();
    }

    bool unique() const noexcept {
        return block_ == nullptr ? true : block_->use_count() == 1;
    }

    T *get() const { return ptr_; }

public:
    T *ptr_;
    SPBlock *block_;
};
