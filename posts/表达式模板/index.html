模板表达式1的定义

模板表达式2是一种将表达式操作转为操作类，用来节省中间变量创建的操作，实现在运行时大小已知的数组上进行高效运算，而模板元编程目标则为在编译期进行计算，从而降低运行时的开销。

template <typename T>
struct SArray {
public:
    explicit SArray(size_t size) : storage_(new T[size]), storage_size_(size) {
        init();
    }

    SArray(SArray<T> const &that)
        : storage_(new T[that.size()]), storage_size_(that.size()) {
        copy(that);
    }

    SArray &operator=(SArray<T> const &that) {
        copy(that);
        return *this;
    }

    ~SArray() { delete[] storage_; }

    T operator[](size_t idx) const {
        assert(idx < size());
        return storage_[idx];
    }

    T &operator[](size_t idx) {
        assert(idx < size());
        return storage_[idx];
    }

    size_t size() const { return storage_size_; }

private:
    void init() {
        for (size_t i = 0; i < size(); i++) {
            storage_[i] = T{};
        }
    }

    void copy(SArray<T> const &that) {
        assert(that.size() == size());
        for (size_t i = 0; i < size(); i++) {
            storage_[i] = that.storage_[i];
        }
    }

    T *storage_;
    size_t storage_size_;
};

template <typename T>
SArray<T> operator*(SArray<T> const &lhs, SArray<T> const &rhs) {
    SArray<T> result(lhs.size());
    for (size_t i = 0; i < lhs.size(); i++) {
        result[i] = lhs[i] * rhs[i];
    }
    return result;
}

template <typename T>
SArray<T> operator+(SArray<T> const &lhs, SArray<T> const &rhs) {
    SArray<T> result(lhs.size());
    for (size_t i = 0; i < lhs.size(); i++) {
        result[i] = lhs[i] + rhs[i];
    }
    return result;
}

template <typename T, typename K>
SArray<T> operator*(SArray<T> const &lhs, K k) {
    SArray<T> result(lhs.size());
    for (size_t i = 0; i < lhs.size(); i++) {
        result[i] = lhs[i] * k;
    }
    return result;
}

template <typename T, typename K>
SArray<T> operator*(K k, SArray<T> const &lhs) {
    return lhs * k;
}

按照计算的优先级，首先 expr1 = 1.2 * x 会在运算过程中构建局部的线性表对象用来存储中间结果，其次是 expr2 = x * y, 最后是上述两个表达式的和 expr1 + expr2.

我们可以看出，计算在 SArray 上的表达式复杂度不仅与线性表的复杂度相关，同时与表达式的复杂度则程度有关，在这个示例中，为了存储中间变量，还额外引入了三次相同大小的线性表维护开销。

那么如何降低这样的开销，令其复杂度能尽可能接近下述操作呢？

for(size_t idx = 0; idx < x.size(); i++) {
    x[i] = 1.2 * x[i] + x[i] * y[i];

// SArray的自加运算符
template<class T>
SArray<T>& SArray<T>::operator+= (SArray<T> const& b)
{
    for (size_t k = 0; k<size(); ++k) {
      (*this)[k] += b[k];
    }
    return *this;
}

// SArray的自乘运算符
template<class T>
SArray<T>& SArray<T>::operator*= (SArray<T> const& b)
{
    for (size_t k = 0; k<size(); ++k) {
      (*this)[k] *= b[k];
    }
    return *this;
}
// 针对放大倍数的自乘运算符
template<class T>
SArray<T>& SArray<T>::operator*= (T const& s)
{
    for (size_t k = 0; k<size(); ++k) {
      (*this)[k] *= s;
    }
    return *this;
}

template<typename T, typename T1, typename T2, typename Op>
SArray<T> calc(SArray<T1> const &lhs, SArray<T2> const &rhs, Op op) {
    SArray<T> result(lhs.size());
    for(size_t i = 0; i < lhs.size(); i++) {
        result[i] = op(lhs[i], rhs[i]);
    }
    return result;
}

int main()
{
    SArray<double> x(1000), y(1000);
    // 中间构造了三次局部 SArray
    // x = 1.2 * x + x * y;
    // 中间构造了一次局部 SArray
    x = calc<double>(x, y, [](auto l, auto r) {
        return 1.2 * l + l * r;
    });
    return 0;
}

与上述重载内建运算-赋值操作类似。

表达式模板

为了实现最开始的目标，我们需要使用一种方法，能够在编译时得到完整的表达式，同时只有实际需要赋值的操作时才开始运算，延迟计算的时间。

也就是说可以将上述的表达式转换为：

1
2
3

1.2 * l + l * r // 这一表达式可以隐式转换成类型

struct Add<Mul<Scalar,Array<double>>, Mul<Array<double>, Array<double>>>;

当我们需要线性表中具体的某一位结果才开始进行计算，这样就不需要中间变量，同时性能不会明显下降。

借助编译器，我们可以直接获得该表达式的语法树，所以这部分可以不用担心。而对于具体的操作类型，也即 Add,Mul，因为实际分配内存的线性表生命周期一定等于或大于表达式所在行，所以，构建过程可以递归地传递引用，以间接访问实际线性表。

template <typename>
struct AScalar;

template <typename T>
struct ATraits {
    // using ExprRef = std::add_const_t<std::add_lvalue_reference_t<T>>;
    using ExprRef = std::add_lvalue_reference_t<std::add_const_t<T>>;
    // using ExprRef = A const &;
};

template <typename T>
struct ATraits<AScalar<T>> {
    using ExprRef = AScalar<T>;
};

template <typename T, typename OP1, typename OP2>
struct AAdd {
public:
    AAdd(OP1 const &a, OP2 const &b) : op1(a), op2(b) {}

    T operator[](size_t idx) const { return op1[idx] + op2[idx]; }

    size_t size() const {
        assert(op1.size() == 0 or op2.size() == 0 or op1.size() == op2.size());
        return op1.size() == 0 ? op2.size() : op1.size();
    }

private:
    ATraits<OP1>::ExprRef op1;
    ATraits<OP2>::ExprRef op2;
};

template <typename T, typename OP1, typename OP2>
struct AMul {
public:
    AMul(OP1 const &a, OP2 const &b) : op1(a), op2(b) {}

    T operator[](size_t idx) const { return op1[idx] * op2[idx]; }

    size_t size() const {
        assert(op1.size() == 0 or op2.size() == 0 or op1.size() == op2.size());
        return op1.size() == 0 ? op2.size() : op1.size();
    }

private:
    ATraits<OP1>::ExprRef op1;
    ATraits<OP2>::ExprRef op2;
};

template <typename T>
struct AScalar {
public:
    AScalar(T const &a) : val(a) {}

    T operator[](size_t) const { return val; }

    size_t size() const { return 0; }

private:
    const T &val;
};

这里提供基本的实现，通过 operator[] 下标访问时计算，同时可以看到，对于标量这一生命周期可能短于表达式所在行的情况，则不采用引用传递的方式构建，而是直接赋值构造，这一点可以通过萃取类得出。

template <typename T, typename Rep = SArray<T>>
struct Array {
public:
    explicit Array(size_t size) : expr_rep(size) {}

    Array(Array const &that) : expr_rep(that.expr_rep) {}

    Array(Rep const &that) : expr_rep(that) {}

    Array &operator=(Array const &that) {
        assert(size() == that.size());
        for (size_t i = 0; i < that.size(); i++) {
            expr_rep[i] = that[i];
        }
        return *this;
    }

    template <typename T1, typename Rep1>
    Array &operator=(Array<T1, Rep1> const &that) {
        assert(size() == that.size());
        for (size_t i = 0; i < that.size(); i++) {
            expr_rep[i] = that[i];
        }
        return *this;
    }

    size_t size() const noexcept { return expr_rep.size(); }

    T operator[](size_t idx) const {
        assert(idx < size());
        return expr_rep[idx];
    }

    T &operator[](size_t idx) {
        assert(idx < size());
        return expr_rep[idx];
    }

    Rep const &expr() const noexcept { return expr_rep; }

    Rep &expr() noexcept { return expr_rep; }

private:
    Rep expr_rep;
};

template <typename T, typename R1, typename R2>
Array<T, AAdd<T, R1, R2>> operator+(Array<T, R1> const &expr1,
                                    Array<T, R2> const &expr2) {
    return Array<T, AAdd<T, R1, R2>>(
        AAdd<T, R1, R2>(expr1.expr(), expr2.expr()));
}

template <typename T, typename R1, typename R2>
Array<T, AMul<T, R1, R2>> operator*(Array<T, R1> const &expr1,
                                    Array<T, R2> const &expr2) {
    return Array<T, AMul<T, R1, R2>>(
        AMul<T, R1, R2>(expr1.expr(), expr2.expr()));
}

template <typename T, typename R2>
Array<T, AMul<T, AScalar<T>, R2>> operator*(T const &val,
                                            Array<T, R2> const &expr2) {
    return Array<T, AMul<T, AScalar<T>, R2>>(
        AMul<T, AScalar<T>, R2>(AScalar<T>(val), expr2.expr()));
}

template <typename T, typename R1>
Array<T, AMul<T, R1, AScalar<T>>> operator*(Array<T, R1> const &expr1,
                                            T const &val) {
    return Array<T, AMul<T, R1, AScalar<T>>>(
        AMul<T, R1, AScalar<T>>(expr1.expr(), AScalar<T>(val)));
}

template <typename T, typename R2>
Array<T, AAdd<T, AScalar<T>, R2>> operator+(T const &val,
                                            Array<T, R2> const &expr2) {
    return Array<T, AAdd<T, AScalar<T>, R2>>(
        AAdd<T, AScalar<T>, R2>(AScalar<T>(val), expr2.expr()));
}

template <typename T, typename R1>
Array<T, AAdd<T, R1, AScalar<T>>> operator+(Array<T, R1> const &expr1,
                                            T const &val) {
    return Array<T, AAdd<T, R1, AScalar<T>>>(
        AAdd<T, R1, AScalar<T>>(expr1.expr(), AScalar<T>(val)));
}
